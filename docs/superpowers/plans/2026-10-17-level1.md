> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Turn the start island into a full spearfishing level: catch fish and crabs (with sizes), sell them on Coco's dock scale, buy gear at a shop wall, gamble at a paid chest, and kill the shark boss (with a health bar) to open the way to island 2.

**Architecture:** This follows the existing node/inspector-first patterns. Each new thing is a scene with `@export`s:
- New scenes: `SellScale`, `Coco`, `ShopItem`, `Crab`, `CrabSpawner`.
- The interact system (`get_prompt()` / `interact(player)`, interact ray mask 16) and the combat system (`take_damage(amount, at)`, attack cast mask 32) are reused unchanged.
- Money and per-slot weights live on `Player`.
- Sell values and size rules live on `ItemData`.

**Tech Stack:** Godot 4.7.2 (Forward+, Jolt), GDScript. Headless test scripts run with the console exe.

**Spec:** `docs/superpowers/specs/2026-10-17-level1-design.md`

## Global Constraints
- **Godot executable:** `C:/Users/erlen/Downloads/Godot_v4.7.2-stable_win64.exe/Godot_v4.7.2-stable_win64_console.exe`
- **Run a test:** `"$GODOT" --headless --path . -s tests/<name>.gd`. A test prints `PASS <name>` and quits with code 0, or fails on an `assert`.
- **Inspector-first:** tunables are `@export`s, and layout lives in `.tscn` files. Keep scripts small.
- **Physics layers:** 1 world, 2 boat_hull, 3 boat_deck (4), 4 player (8), 5 interactables (16), 6 creatures (32).
  - Things you press E on: `collision_layer = 17`.
  - Things you hit: layer 32.
- **ASCII only in `.gd` files.** When editing from Python, use `encoding="utf-8"`.
- **No AI-generated 3D models or audio.** Placeholders are Godot primitive meshes.
- **Balance numbers:**
  - Damage: fist 2, rock 5, spear 20. Shark health 200.
  - Shop: spear $40, bottle $8. Gamble $15.
- **Hotbar:** 5 slots.
- **Controls:** Q drops toward the crosshair, E sells, E buys.

## Review Focus
1. **Selling tools by accident:** a tool with `price = 0` dropped on the tray stays there and doesn't count toward the total. (Task 2 test.)
2. **Buying with a full hotbar:** the item drops at your feet, and money is still spent exactly once. (Task 3 test.)
3. **Pressing E on the gamble chest without money, or while it's open:** no charge, no loot. (Task 5 test.)
4. **Dying mid-fight:** the next shark comes back at full health. Once the shark is defeated, deep water never spawns another. (Task 4 test.)
5. **Weight surviving pick up and drop:** a 3 kg fish picked up and dropped is still 3 kg and still looks big. (Task 1 test.)

```gdscript
@export_group("Selling")
## What the scale pays: per item, or per kg when sold_by_weight. 0 = can't be sold.
@export var price := 0
@export var sold_by_weight := false
## Caught creatures come in sizes (kg); (0, 0) = no size.
@export var weight_range := Vector2.ZERO
## The weight at which the model is its normal size.
@export var base_weight := 1.0


## A random size: mostly around the middle, sometimes small, rarely big.
func roll_weight(rng: RandomNumberGenerator) -> float:
	if weight_range == Vector2.ZERO:
		return 0.0
	var t := clampf(rng.randfn(0.4, 0.18), 0.0, 1.0)
	return snappedf(lerpf(weight_range.x, weight_range.y, t), 0.1)


## Model scale for a weight (mass grows with volume, so scale is the cube root).
func size_scale(weight: float) -> float:
	return pow(weight / base_weight, 1.0 / 3.0) if weight > 0.0 else 1.0


func value(weight: float) -> int:
	return roundi(price * weight) if sold_by_weight else price
```

```gdscript
## A lob that lands where the crosshair points (up to 4 m away), so you can aim drops onto the scale.
func _toss_velocity(from: Vector3) -> Vector3:
	var space := get_world_3d().direct_space_state
	var aim_from := _camera.global_position
	var aim_to := aim_from - _camera.global_basis.z * 4.0
	var query := PhysicsRayQueryParameters3D.create(aim_from, aim_to, 1 | 4 | 16, [get_rid()])
	var hit := space.intersect_ray(query)
	var target: Vector3 = hit.position if hit else aim_to
	var t := 0.45  # flight time
	return (target - from) / t - get_gravity() * t * 0.5
```

- [ ] **Step 4: `scripts/shop/sell_scale.gd`**

```gdscript
class_name SellScale
extends StaticBody3D
## A dock scale. Drop catch onto the tray (Q); the sign shows the total weight and price; press E
## on the scale to sell everything on it. Items that can't be sold (price 0, e.g. tools) stay.

## Finds the items lying on the tray.
@export var tray_area: Area3D
@export var sign_label: Label3D
@export var coco: Coco
## A sale pays this much or more per item when it counts as a big catch ("chonker" lines).
@export var big_value := 15

var _sold := 0.0  # seconds the "+$" text is still shown


func _process(delta: float) -> void:
	_sold -= delta
	if _sold > 0.0:
		return
	var items := _on_tray()
	var kg := 0.0
	for it in items:
		kg += it.weight
	sign_label.text = "%.1f kg\n$%d" % [kg, total()] if items else "Drop catch here\n[Q]"


func _on_tray() -> Array[WorldItem]:
	var out: Array[WorldItem] = []
	for body in tray_area.get_overlapping_bodies():
		if body is WorldItem and body.item.price > 0:
			out.append(body)
	return out


func total() -> int:
	var sum := 0
	for it in _on_tray():
		sum += it.item.value(it.weight)
	return sum


func get_prompt() -> String:
	var sum := total()
	return "E  sell $%d" % sum if sum > 0 else ""


func interact(player: Node) -> void:
	var sum := total()
	if sum == 0:
		return
	var big := false
	for it in _on_tray():
		big = big or it.item.value(it.weight) >= big_value
		it.queue_free()
	player.add_money(sum)
	sign_label.text = "+$%d" % sum
	_sold = 1.5
	if coco:
		coco.say(big)
```

```gdscript
class_name Crab
extends Node3D
## A beach crab: wanders along the sand, scuttles away sideways when the player comes close,
## and leaves `item` (with its size) behind when killed. Sticks to the terrain height.

@export var item: ItemData
@export var health := 6.0
@export var speed := 0.8
@export var flee_speed := 3.5
## Flees when the player is closer than this (m).
@export var scare_distance := 4.0
## Stays on ground between these heights (the beach).
@export var min_height := 0.7
@export var max_height := 2.5

var terrain: Terrain
var player: Node3D
var weight := 0.0

var _target := Vector3.ZERO
var _wait := 0.0


func _ready() -> void:
	if item:
		weight = item.roll_weight(RandomNumberGenerator.new())
		$Model.scale *= item.size_scale(weight)
	_target = global_position


func _physics_process(delta: float) -> void:
	var move_speed := speed
	if player and global_position.distance_to(player.global_position) < scare_distance:
		var away := (global_position - player.global_position) * Vector3(1, 0, 1)
		_target = global_position + away.normalized() * 3.0
		move_speed = flee_speed
	elif global_position.distance_to(_target) < 0.2:
		_wait -= delta
		if _wait <= 0.0:
			_pick_target()
		return
	var step := (_target - global_position) * Vector3(1, 0, 1)
	if step.length() < 0.01:
		return
	var next := global_position + step.normalized() * minf(move_speed * delta, step.length())
	var h := terrain.get_floor_height(next.x, next.z)
	if h < min_height or h > max_height:
		_pick_target()  # don't walk into the sea or up the hill
		return
	global_position = Vector3(next.x, h, next.z)
	# Crabs walk sideways: face across the direction of travel.
	look_at(global_position + step.normalized().cross(Vector3.UP), Vector3.UP)


func _pick_target() -> void:
	_wait = randf_range(0.5, 2.5)
	var a := randf() * TAU
	_target = global_position + Vector3(cos(a), 0, sin(a)) * randf_range(1.0, 4.0)


func take_damage(amount: float, _at := Vector3.ZERO) -> void:
	health -= amount
	if health > 0.0:
		_wait = 0.0
		return
	if item:
		var dropped: WorldItem = load(item.scene_path).instantiate()
		dropped.weight = weight
		var world := get_tree().current_scene if get_tree().current_scene else get_tree().root.get_child(-1)
		world.add_child(dropped)  # tests run without a current_scene: main is the root's last child
		dropped.global_position = global_position + Vector3.UP * 0.3
	queue_free()
```

```gdscript
class_name CrabSpawner
extends Node3D
## Puts `count` crabs on the beach of `island` (ground between the crab's min/max height).

@export var terrain: Terrain
@export var island: IslandShape
@export var player: Node3D
@export var crab_scene: PackedScene
@export var count := 6
@export var random_seed := 3


func _ready() -> void:
	var rng := RandomNumberGenerator.new()
	rng.seed = random_seed
	var placed := 0
	for attempt in count * 30:
		if placed == count:
			break
		var a := rng.randf() * TAU
		var r := island.radius * rng.randf_range(0.5, 1.1)
		var x := island.position.x + cos(a) * r
		var z := island.position.z + sin(a) * r
		var h := terrain.get_floor_height(x, z)
		var crab: Crab = crab_scene.instantiate()
		if h < crab.min_height or h > crab.max_height:
			crab.free()
			continue
		crab.terrain = terrain
		crab.player = player
		add_child(crab)
		crab.global_position = Vector3(x, h, z)
		placed += 1
```
